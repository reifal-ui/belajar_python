# Ini adalah todolist API menggunakan FLASK dan juga bisa di akses oleh Remix.js

## End Point
-- GET -> Untuk mengambil semua to-do yang sudah ada
-- POST -> Untuk menambah to-do baru
-- DELETE -> Untuk menghapus to-do 

## Untuk menjalankan API
# (Di dalam folder 06_api)
python -m venv venv
source venv/bin/activate  # atau venv\Scripts\activate di Windows
pip install -r requirements.txt
python app.py
